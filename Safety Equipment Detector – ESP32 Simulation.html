void loop() {
  bool motion = digitalRead(PIR);      // G27
  bool helmet = digitalRead(HELMET);   // G26
  bool vest   = digitalRead(VEST);     // G25
  showStatus(motion, helmet, vest);    // OLED
  if (motion && (!helmet || !vest)) {
    digitalWrite(BUZZER, HIGH);        // alert
  } else {
    digitalWrite(BUZZER, LOW);
  }
  delay(100);
}

How it works

  1. A person enters and the PIR sensor detects motion.
  2. The reed switch closes if a magnet in the helmet is near.
  3. The IR sensor sees the reflective tape on the vest.
  4. The ESP32 shows all three readings on the OLED.
  5. If anything is missing, the buzzer sounds and the missing item is highlighted.
